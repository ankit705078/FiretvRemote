Swipe Touchpad Vector Math (`Δx`, `Δy`)

`firetv_bridge.py` now hosts `firetv_remote.html` over `0.0.0.0:8765` so you can open `http://<your-pc-ip>:8765` on your phone browser and control your physical Fire TV Stick!

1. Run on Phone via Wi-Fi ADB (2 Minutes)
  1. Enable Settings > My Fire TV > Developer Options > ADB Debugging (Amazon ADB Guide ↗).
  2. Download `firecontrol_complete_app.zip` and run: python3 firetv_bridge.py <FIRE_TV_IP>:5555.
  3. Open the printed `http://<LAN_IP>:8765` URL on your phone browser (or in Android Termux) and tap Add to Home Screen!
2. Run Offline via Bluetooth HID (No Wi-Fi)
  1. On Fire TV, open Settings > Controllers & Bluetooth Devices > Other Bluetooth Devices.
  2. Select Add Bluetooth Devices to put Fire TV into pairing mode.
  3. Use `AndroidBluetoothHidHelper.kt` (included in the zip) with Android's BluetoothHidDevice API ↗ to pair your phone directly as a Bluetooth HID remote!
# firetv_bridge.py (Complete Phone + Fire TV Server)
# Hosts firetv_remote.html for your phone AND forwards ADB commands to Fire TV!
import os, socket, subprocess, sys, urllib.parse
from http.server import BaseHTTPRequestHandler, HTTPServer

FIRETV_TARGET = sys.argv[1] if len(sys.argv) > 1 else "192.168.1.45:5555"
subprocess.run(["adb", "connect", FIRETV_TARGET])

class BridgeHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path in ("/", "/index.html", "/firetv_remote.html"):
            self.send_response(200)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.end_headers()
            with open(os.path.join(os.path.dirname(__file__), "firetv_remote.html"), "rb") as f:
                self.wfile.write(f.read())
            return
        if self.path.startswith("/keyevent/"):
            code = self.path.split("/keyevent/")[1]
            subprocess.run(["adb", "-s", FIRETV_TARGET, "shell", "input", "keyevent", code])
        elif self.path.startswith("/macro/"):
            codes = self.path.split("/macro/")[1].split(",")
            subprocess.run(["adb", "-s", FIRETV_TARGET, "shell", "input", "keyevent", *codes])
        elif self.path.startswith("/swipe/"):
            coords = self.path.split("/swipe/")[1].split(",")
            if len(coords) == 4:
                subprocess.run(["adb", "-s", FIRETV_TARGET, "shell", "input", "swipe", *coords, "150"])
        elif self.path.startswith("/text/"):
            raw = urllib.parse.unquote(self.path.split("/text/")[1]).replace(" ", "%s")
            subprocess.run(["adb", "-s", FIRETV_TARGET, "shell", "input", "text", raw])
        elif self.path.startswith("/launch/"):
            pkg = self.path.split("/launch/")[1]
            subprocess.run(["adb", "-s", FIRETV_TARGET, "shell", "monkey", "-p", pkg, "-c", "android.intent.category.LAUNCHER", "1"])
        self.send_response(200)
        self.send_header("Access-Control-Allow-Origin", "*")
        self.end_headers()
        self.wfile.write(b"OK")

lan_ip = socket.gethostbyname(socket.gethostname())
print(f"Open on your Phone Browser: http://{lan_ip}:8765  (Target Fire TV: {FIRETV_TARGET})")
HTTPServer(("0.0.0.0", 8765), BridgeHandler).serve_forever()
